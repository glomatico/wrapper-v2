cmake_minimum_required(VERSION 3.20)
project(wrapper_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_MakeAvailable(json)
add_library(worker_common native/protocol.cpp native/sha256.cpp)
target_include_directories(worker_common PUBLIC native)
target_link_libraries(worker_common PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(worker_common PRIVATE -Wall -Wextra -Werror)
enable_testing()
if(NOT ANDROID)
  add_executable(worker-launch native/launcher.cpp)
  target_compile_options(worker-launch PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(common_tests tests/common_tests.cpp)
target_link_libraries(common_tests PRIVATE worker_common)
add_test(NAME common COMMAND common_tests)
if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "x86_64" OR NOT ANDROID_PLATFORM_LEVEL EQUAL 21)
    message(FATAL_ERROR "Only Android API 21 x86_64 is validated")
  endif()
  set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
  set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
  set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(dobby GIT_REPOSITORY https://github.com/BepInEx/Dobby.git GIT_TAG 9d271803881f467aabfc24593c7dcaeb710fe861)
  FetchContent_MakeAvailable(dobby)
  add_executable(apple-music-worker native/main.cpp native/runtime.cpp native/session.cpp native/playback.cpp native/context.cpp native/capture_x86_64.cpp)
  target_link_libraries(apple-music-worker PRIVATE worker_common dobby dl log)
  target_compile_options(apple-music-worker PRIVATE -Wall -Wextra -Werror)
  target_link_options(apple-music-worker PRIVATE -Wl,--hash-style=both,-rpath,/system/lib64)
endif()
